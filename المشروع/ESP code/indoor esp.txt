#include <BLEDevice.h>
#include <BLEUtils.h>
#include <BLEServer.h>

#define DEVICE_NAME "ESP32_1"
#define SERVICE_UUID "12345678-1234-1234-1234-1234567890ab"

void setup() {
  Serial.begin(115200);

  // 1. تهيئة الجهاز
  BLEDevice::init(DEVICE_NAME);
  
  // [تعديل]: تثبيت قوة الإرسال (TX Power) على أعلى قيمة ثابتة (+9 dBm) لتقليل تذبذب الـ RSSI
  esp_ble_tx_power_set(ESP_BLE_PWR_TYPE_ADV, ESP_PWR_LVL_P9);
  
  BLEServer *pServer = BLEDevice::createServer();

  // 2. إنشاء الخدمة (Service)
  BLEService *pService = pServer->createService(SERVICE_UUID);
  pService->start();

  // 3. إعداد الإعلان (Advertising) ثابت القيمة والزمن
  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  
  BLEAdvertisementData oAdvertisementData;
  oAdvertisementData.setFlags(0x06); // بث عام مستمر
  oAdvertisementData.setCompleteServices(BLEUUID(SERVICE_UUID));
  oAdvertisementData.setName(DEVICE_NAME);
  
  // [تعديل]: تضمين قيمة الـ TX Power يدوياً (+9 dBm تعادل 0x09) جوه الـ advertising packet
  oAdvertisementData.addData(String((char)0x02) + String((char)0x0A) + String((char)0x09)); 
  
  pAdvertising->setAdvertisementData(oAdvertisementData);
  pAdvertising->setScanResponse(true);

  // [تعديل]: تثبيت زمن الإرسال (Advertising Interval) بدقة على 100ms لمنع التغير الديناميكي
  // 160 * 0.625ms = 100ms ثابتة تماماً
  pAdvertising->setMinInterval(160); 
  pAdvertising->setMaxInterval(160); 

  pAdvertising->setMinPreferred(0x06);  
  pAdvertising->setMinPreferred(0x12);

  // 4. بدء البث
  BLEDevice::startAdvertising();
  Serial.println("✅ ESP32 Beacon Configured & Advertising Stably!");
}

void loop() {
  delay(2000);
}
